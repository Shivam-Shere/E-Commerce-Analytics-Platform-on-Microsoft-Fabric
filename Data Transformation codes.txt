Table: "bronze_products"


Sample data transformation code:
 
# Data Transformation:
 
# Create all calculated columns:
 
df_transformed = (
    df
 
    # 1. Create product Size Category: categorize products based on calculated volume
    .withColumn(
    "product_size_category",
    when(col("product_volume_cm3") < 1000, "Small")
    .when(col("product_volume_cm3") < 10000, "Medium")
    .when(col("product_volume_cm3") < 50000, "Large")
    .otherwise("Oversized")
    )
 

    # 2. Product Weight Category: Segment Products by weight
    .withColumn(
    "product_weight_category",
    when(col("product_weight_g") < 500, "Lightweight")
    .when(col("product_weight_g") < 2000, "Medium Weight")
    .when(col("product_weight_g") < 5000, "Heavy")
    .otherwise("Very Heavy")
    )
 

    # 3. Product Description Quality: Classify product based on product length
    .withColumn(
    "product_description_quality",
    when(col("product_description_lenght") < 200, "Poor")
    .when(col("product_description_lenght") < 600, "Average")
    .otherwise("Good")
    )

 
    # 4. Product Photo Category: Classify products based on no.of photos.
    .withColumn(
    "product_photo_category",
    when(col("product_photos_qty") <= 1, "Low")
    .when(col("product_photos_qty") <= 3, "Medium")
    .otherwise("High")
    )

 
    # 5. Product Volume (cm3): Calculate physical volume of product [Volume = Length × Width × Height]

    .withColumn(
    "product_volume_cm3",
    col("product_length_cm") *
    col("product_width_cm") *
    col("product_height_cm")
    ) 

)
 
 
# Verify data
df_transformed.select(
    "product_size_category",
    "product_weight_category",
    "product_description_quality",
    "product_photo_category",
    "product_volume_cm3"
    
).show(10, truncate=False)

    .when(col("seller_state").isin("AL","BA","CE","MA","PB","PE","PI","RN","SE"),"Northeast")
    .when(col("seller_state").isin("DF","GO","MT","MS"),"Central-West")
    .when(col("seller_state").isin("ES","MG","RJ","SP"),"Southeast")
    .when(col("seller_state").isin("PR","RS","SC"),"South")
    .otherwise("Unknown")
    ) 


    # 2. Seller Full Location: Combine City and State Name.

    .withColumn(
    "seller_full_location",
    concat_ws(", ", col("seller_city"), col("seller_state_name"))
    ) 


 
    # 3. Seller Location Key: Create a reusable location identifier. {e.g., Curitiba - PR}
    
    .withColumn(
    "seller_location_key",
    concat_ws(
        " - ",
        col("seller_city"),
        col("seller_state"))
    )

    .withColumn(
    "seller_zip_prefix_length",
    length(col("seller_zip_code_prefix").cast("string"))
    )


)
