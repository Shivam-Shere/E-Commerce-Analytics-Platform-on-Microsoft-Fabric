Page-1: Overview Analysis 

1. Total Orders:

a. Total Orders= DISTINCTCOUNT(fact_orders[order_id])

b. Total Orders % =
DIVIDE(
    [Total Orders],
    CALCULATE([Total Orders], ALL(dim_date)),
    0
)

c. PY Total Orders =
CALCULATE(
    [Total Orders],
    SAMEPERIODLASTYEAR(dim_date[date])
)

d. YoY Total Orders =
IF(
    ISBLANK([PY Total Orders]),
    BLANK(),
    [Total Orders] - [PY Total Orders]
)

e. YoY % Total Orders =
VAR PrevYear =
COALESCE([PY Total Orders],0)

RETURN
IF(
    PrevYear=0,
    0,
    DIVIDE([YoY Total Orders],PrevYear,0)
)
